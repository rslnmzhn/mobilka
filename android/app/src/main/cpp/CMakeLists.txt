cmake_minimum_required(VERSION 3.22)
project(mobilka_documents_android LANGUAGES CXX)

if(NOT DEFINED DOCUMENTS_PROVISION_ROOT)
  message(FATAL_ERROR "Run native/documents/provision-android.ps1 before building Android")
endif()
set(abi_root "${DOCUMENTS_PROVISION_ROOT}/${ANDROID_ABI}")
if(NOT EXISTS "${abi_root}/provision.cmake")
  message(FATAL_ERROR "Missing verified document worker provision for ${ANDROID_ABI}")
endif()
include("${abi_root}/provision.cmake")
set(DOCUMENTS_PROVISIONING_REVIEWED ON)
set(DOCUMENTS_ENG_DATA "${DOCUMENTS_PROVISION_ROOT}/assets/eng.traineddata")
set(DOCUMENTS_RUS_DATA "${DOCUMENTS_PROVISION_ROOT}/assets/rus.traineddata")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../../native/documents"
  "${CMAKE_CURRENT_BINARY_DIR}/documents")
add_library(mobilka_documents_jni SHARED document_worker_jni.cpp)
target_compile_features(mobilka_documents_jni PRIVATE cxx_std_17)
target_link_libraries(mobilka_documents_jni PRIVATE mobilka_document_engine log)
