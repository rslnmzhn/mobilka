cmake_minimum_required(VERSION 3.22)
project(mobilka_documents LANGUAGES CXX)

include(dependencies.cmake)
add_library(mobilka_document_engine STATIC
  document_engine.cpp document_pdf.cpp document_ocr.cpp)
target_compile_features(mobilka_document_engine PUBLIC cxx_std_17)
if(WIN32)
  target_compile_definitions(mobilka_document_engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
    _ITERATOR_DEBUG_LEVEL=0 _HAS_ITERATOR_DEBUGGING=0)
  set_property(TARGET mobilka_document_engine PROPERTY
    MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  target_compile_options(mobilka_document_engine PRIVATE /MD)
endif()
set_target_properties(mobilka_document_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mobilka_document_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mobilka_document_engine PUBLIC
  documents_PDFIUM documents_TESSERACT documents_LEPTONICA)
if(WIN32 AND NOT DOCUMENTS_RUNTIME_FILES)
  message(FATAL_ERROR "Verified Windows document worker runtime files are required")
endif()

option(DOCUMENTS_BUILD_TESTS "Build isolated native engine tests" OFF)
if(DOCUMENTS_BUILD_TESTS)
  enable_testing()
  add_executable(document_engine_test tests/document_engine_test.cpp)
  target_link_libraries(document_engine_test PRIVATE mobilka_document_engine)
  if(WIN32)
    target_compile_definitions(document_engine_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(document_engine_test PRIVATE gdi32 user32)
  endif()
  foreach(language ENG RUS)
    file(TO_CMAKE_PATH "${DOCUMENTS_${language}_DATA}" test_data_path)
    string(REPLACE "\"" "\\\"" test_data_path "${test_data_path}")
    target_compile_definitions(document_engine_test PRIVATE
      "DOCUMENTS_${language}_DATA=\"${test_data_path}\"")
  endforeach()
  add_test(NAME document_engine_test COMMAND document_engine_test)
  add_executable(android_protocol_test tests/android_protocol_test.cpp)
  target_link_libraries(android_protocol_test PRIVATE mobilka_document_engine)
  add_test(NAME android_protocol_test COMMAND android_protocol_test)
endif()
